Add The DefaultConnection To the Prograam.cs File For Run The Server to Access The Db


using Microsoft.EntityFrameworkCore;
using WebApp.Db;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();

// Swagger
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();
builder.Services.AddDbContext<ApplicationDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));
var app = builder.Build();           ///////////// DbContext CreatedAfter thissss.

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.UseAuthorization();

app.MapControllers();

app.Run();

Create API


	=> Create a Controller - Create a folder "Controller" to Create a Controller name same as model name with Controller


using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Mvc;
using WebApp.Db;
using WebApp.Model;

namespace WebApp.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class StudController : ControllerBase
    {
        private readonly ApplicationDbContext _db;
        public StudController(ApplicationDbContext Db)
        {
            _db = Db;
        }
        [HttpPost]
        [Route("Create")]
        public IActionResult Create([FromBody] Stud s1)
        {
            _db.Studs.Add(s1);
            _db.SaveChanges();
            return Ok(s1);
        }
        [HttpGet]
        [Route("GetAllStud")]
        public IActionResult GetAll()
        {
            List<Stud> Stud = new List<Stud>();
            Stud = _db.Studs.ToList();
            return Ok(Stud);
        }
        [HttpGet]
        [Route("GetById")]
        public IActionResult GetById(int id)
        {
            Stud s1 = _db.Studs.FirstOrDefault(x => x.id == id);
            return Ok(s1);
        }
        [HttpPut]
        [Route("Update")]
        public IActionResult Update([FromBody] Stud s1)
        {
            _db.Update(s1);
            _db.SaveChanges();
            return Ok(s1);
        }
        [HttpDelete]
        [Route("Delete")]
        public IActionResult DeleteById(int id)
        {
            Stud s1 = _db.Studs.FirstOrDefault(x => x.id == id);
            _db.Remove(s1);
            _db.SaveChanges();
            return Ok(s1);
        }
    }
}  //Example

--------------React------------
-> React is a JavaScript library used to build user interfaces (UI), especially for web applications.
-> React updates the UI automatically when data changes.
=> Reusable Components
-> Instead of writing the same code repeatedly, you create a component once and reuse it.





Component :- Create a Component:-

	function Navbar() {  // create a function to display the navbar on the page
    return(
        <nav>
            <h2>app</h2>
            <button> Log In</button>
        </nav>
    );
    
}
export default Navbar;  //export for use in main file to display to op

then import these Navbar file in App.jsx

Import:-
	
import './App.css'
import Navbar from "./Navbar";
function App() {
  return (
    <>
      <Navbar />
    </>
  );
}

export default App

import HTML from "./assets/htcsjs.jpg";

function Course(props) {
    return (
        <div className="card">
            <img src={props.image} alt="" />
            <h3>{props.name}</h3>
            <p>{props.price}</p>
            <p>{props.rating}</p>
        </div>
    );
}

// Default props
Course.defaultProps = {
    name: "No Course",
    price: "$99",
    image: HTML,
    rating: "0⭐"
};
